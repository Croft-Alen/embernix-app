function discountLabel(coupon: Coupon): string {
  if (coupon.discount_type === "percentage") {
    return `${coupon.amount}%`;
  }

  const currency = coupon.currency ?? "USD";

  try {
    const formatter = new Intl.NumberFormat("en-US", {
      style: "currency",
      currency,
    });

    const decimals =
      formatter.resolvedOptions().maximumFractionDigits ?? 2;

    return formatter.format(
      coupon.amount / 10 ** decimals
    );
  } catch {
    return `${coupon.amount} ${currency}`;
  }
}

function dateLabel(value: string | null): string {
  if (!value) return "Never";

  const date = new Date(value);

  return Number.isNaN(date.getTime())
    ? "Invalid date"
    : date.toLocaleString("en-US", {
        dateStyle: "medium",
        timeStyle: "short",
      });
}

function expired(value: string | null): boolean {
  if (!value) return false;

  const time = new Date(value).getTime();

  return !Number.isFinite(time) || time <= Date.now();
}

function getStatus(coupon: Coupon) {
  if (!coupon.active) {
    return {
      text: "Inactive",
      className: "bg-gray-100 text-gray-600",
      Icon: XCircle,
    };
  }

  if (!coupon.creem_discount_id) {
    return {
      text: "Needs sync",
      className: "bg-amber-50 text-amber-700",
      Icon: AlertTriangle,
    };
  }

  if (expired(coupon.expires_at)) {
    return {
      text: "Expired",
      className: "bg-orange-50 text-orange-700",
      Icon: Clock3,
    };
  }

  return {
    text: "Active locally",
    className: "bg-green-50 text-green-700",
    Icon: CheckCircle2,
  };
}

"use server";

import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

import {
  createCreemDiscount,
  getCreemDiscountByCode,
} from "@/lib/creem/server";
import { createAdminClient } from "@/lib/supabase/admin";
import { createClient } from "@/lib/supabase/server";

type DiscountType = "percentage" | "flat";

type CouponData = {
  code: string;
  description: string | null;
  discount_type: DiscountType;
  amount: number;
  currency: string | null;
  active: boolean;
  expires_at: string | null;
  usage_limit: number | null;
};

type StoredCoupon = CouponData & {
  id: string;
  creem_discount_id: string | null;
};

function read(value: FormDataEntryValue | null): string {
  return String(value ?? "").trim();
}

function message(error: unknown): string {
  return error instanceof Error
    ? error.message
    : "Unable to process coupon.";
}

function fail(path: string, reason: string): never {
  redirect(`${path}?${new URLSearchParams({ error: reason })}`);
}

function positiveInteger(value: string): number | null {
  if (!/^\d+$/.test(value)) return null;

  const number = Number(value);

  return Number.isSafeInteger(number) && number > 0
    ? number
    : null;
}

function currencyDecimals(currency: string): number {
  if (!/^[A-Z]{3}$/.test(currency)) {
    throw new Error("Invalid currency.");
  }

  try {
    const decimals =
      new Intl.NumberFormat("en-US", {
        style: "currency",
        currency,
      }).resolvedOptions().maximumFractionDigits ?? 2;

    if (decimals < 0 || decimals > 3) {
      throw new Error("Unsupported currency precision.");
    }

    return decimals;
  } catch {
    throw new Error("Unsupported currency.");
  }
}

function parseFixedAmount(
  raw: string,
  currency: string
): number {
  const decimals = currencyDecimals(currency);

  const pattern =
    decimals === 0
      ? /^(\d+)$/
      : new RegExp(`^(\\d+)(?:\\.(\\d{1,${decimals}}))?$`);

  const match = pattern.exec(raw);

  if (!match) {
    throw new Error("Invalid fixed discount amount.");
  }

  const whole = Number(match[1]);
  const fraction =
    decimals === 0
      ? 0
      : Number((match[2] ?? "").padEnd(decimals, "0"));

  const amount = whole * 10 ** decimals + fraction;

  if (
    !Number.isSafeInteger(amount) ||
    amount <= 0 ||
    amount > 2147483647
  ) {
    throw new Error("Fixed discount is out of range.");
  }

  return amount;
}

async function requireAdmin(): Promise<void> {
  const supabase = await createClient();

  const {
    data: { user },
    error,
  } = await supabase.auth.getUser();

  if (error || !user) {
    redirect("/login");
  }

  const { data: admin, error: adminError } = await supabase
    .from("admin_users")
    .select("user_id")
    .eq("user_id", user.id)
    .maybeSingle();

  if (adminError || !admin) {
    redirect("/dashboard");
  }
}

function parseCoupon(formData: FormData): CouponData {
  const code = read(formData.get("code")).toUpperCase();
  const description = read(formData.get("description")) || null;
  const type = read(formData.get("discountType"));
  const rawAmount = read(formData.get("amount"));
  const rawCurrency = read(formData.get("currency")).toUpperCase();
  const rawExpiry = read(formData.get("expiresAt"));
  const rawLimit = read(formData.get("usageLimit"));

  if (!/^[A-Z0-9]{1,32}$/.test(code)) {
    throw new Error("Coupon code must contain 1-32 letters or numbers.");
  }

  if (description && description.length > 500) {
    throw new Error("Description is too long.");
  }

  if (type !== "percentage" && type !== "flat") {
    throw new Error("Invalid discount type.");
  }

  let amount: number;
  let currency: string | null = null;

  if (type === "percentage") {
    const percentage = positiveInteger(rawAmount);

    if (!percentage || percentage > 100) {
      throw new Error("Percentage must be between 1 and 100.");
    }

    amount = percentage;
  } else {
    currency = rawCurrency;
    amount = parseFixedAmount(rawAmount, currency);
  }

  let usageLimit: number | null = null;

  if (rawLimit) {
    usageLimit = positiveInteger(rawLimit);

    if (!usageLimit || usageLimit > 2147483647) {
      throw new Error("Invalid usage limit.");
    }
  }

  let expiresAt: string | null = null;

  if (rawExpiry) {
    const date = new Date(rawExpiry);

    if (!Number.isFinite(date.getTime())) {
      throw new Error("Invalid expiration date.");
    }

    expiresAt = date.toISOString();
  }

  return {
    code,
    description,
    discount_type: type,
    amount,
    currency,
    active: formData.has("active"),
    expires_at: expiresAt,
    usage_limit: usageLimit,
  };
}

async function loadCoupon(id: string): Promise<StoredCoupon> {
  const admin = createAdminClient();

  const { data, error } = await admin
    .from("coupons")
    .select(
      "id,code,description,discount_type,amount,currency,active,expires_at,usage_limit,creem_discount_id"
    )
    .eq("id", id)
    .maybeSingle();

  if (error || !data) {
    throw new Error("Coupon not found.");
  }

  if (
    data.discount_type !== "percentage" &&
    data.discount_type !== "flat"
  ) {
    throw new Error("Unsupported discount type.");
  }

  return data as StoredCoupon;
}

function equalDates(
  first: string | null,
  second: string | null
): boolean {
  if (!first && !second) return true;
  if (!first || !second) return false;

  return (
    new Date(first).getTime() ===
    new Date(second).getTime()
  );
}

function termsChanged(
  first: CouponData,
  second: CouponData
): boolean {
  return (
    first.code !== second.code ||
    first.discount_type !== second.discount_type ||
    first.amount !== second.amount ||
    first.currency !== second.currency ||
    first.usage_limit !== second.usage_limit ||
    !equalDates(first.expires_at, second.expires_at)
  );
}

async function getProductIds(): Promise<string[]> {
  const admin = createAdminClient();

  const { data, error } = await admin
    .from("products")
    .select("creem_product_id")
    .eq("active", true);

  if (error) {
    throw new Error("Unable to load products.");
  }

  const ids = [
    ...new Set(
      (data ?? [])
        .map((product) => product.creem_product_id)
        .filter(
          (id): id is string =>
            typeof id === "string" &&
            /^prod_[A-Za-z0-9_-]+$/.test(id)
        )
    ),
  ];

  if (ids.length === 0) {
    throw new Error("No eligible Creem products found.");
  }

  return ids;
}

async function ensureRemoteCodeAvailable(
  code: string
): Promise<void> {
  try {
    const existing = await getCreemDiscountByCode(code);

    if (existing.id) {
      throw new Error(
        "This code already exists in Creem. Resolve the existing discount before synchronizing."
      );
    }
  } catch (error) {
    if (
      error instanceof Error &&
      error.message.includes("already exists in Creem")
    ) {
      throw error;
    }

    if (
      error instanceof Error &&
      error.message.includes("(404)")
    ) {
      return;
    }

    throw new Error(
      "Unable to confirm whether this code already exists in Creem."
    );
  }
}

async function createRemote(
  coupon: CouponData
): Promise<string> {
  await ensureRemoteCodeAvailable(coupon.code);

  const products = await getProductIds();

  const discount = await createCreemDiscount({
    name: coupon.code,
    code: coupon.code,
    type:
      coupon.discount_type === "percentage"
        ? "percentage"
        : "fixed",
    ...(coupon.discount_type === "percentage"
      ? { percentage: coupon.amount }
      : {
          amount: coupon.amount,
          currency: coupon.currency ?? "USD",
        }),
    maxRedemptions: coupon.usage_limit,
    expiryDate: coupon.expires_at,
    productIds: products,
  });

  if (
    !/^dis_[A-Za-z0-9_-]+$/.test(discount.id) ||
    discount.code.toUpperCase() !== coupon.code
  ) {
    throw new Error("Creem returned an unexpected discount.");
  }

  return discount.id;
}

async function deleteRemote(id: string): Promise<void> {
  if (!/^dis_[A-Za-z0-9_-]+$/.test(id)) {
    throw new Error("Invalid Creem discount ID.");
  }

  const key = process.env.CREEM_API_KEY;
  const environment = process.env.CREEM_ENVIRONMENT;

  if (
    !key ||
    (environment !== "test" && environment !== "production")
  ) {
    throw new Error("Creem configuration is incomplete.");
  }

  const base =
    environment === "test"
      ? "https://test-api.creem.io"
      : "https://api.creem.io";

  const response = await fetch(
    `${base}/v1/discounts/${encodeURIComponent(id)}/delete`,
    {
      method: "DELETE",
      headers: {
        "x-api-key": key,
      },
      cache: "no-store",
      signal: AbortSignal.timeout(15000),
    }
  );

  if (!response.ok && response.status !== 404) {
    throw new Error(
      `Creem discount deletion failed (${response.status}).`
    );
  }
}

async function cleanupRemote(id: string): Promise<void> {
  try {
    await deleteRemote(id);
  } catch (error) {
    console.error("Manual Creem cleanup required:", id, error);

    throw new Error(
      "The database operation failed and the remote discount requires review."
    );
  }
}

async function ensureNoActiveCheckouts(
  code: string
): Promise<void> {
  const admin = createAdminClient();

  const { data, error } = await admin
    .from("orders")
    .select("id")
    .eq("payment_provider", "creem")
    .eq("status", "pending")
    .eq("payment_status", "unpaid")
    .eq("creem_discount_code", code)
    .in("creem_checkout_state", ["creating", "ready"])
    .limit(1);

  if (error) {
    throw new Error("Unable to verify pending checkouts.");
  }

  if (data && data.length > 0) {
    throw new Error(
      "This coupon has an active unpaid checkout. Resolve that checkout before disabling the discount."
    );
  }
}

export async function createCoupon(
  formData: FormData
): Promise<void> {
  await requireAdmin();

  const path = "/admin/coupons/new";

  let coupon: CouponData;

  try {
    coupon = parseCoupon(formData);
  } catch (error) {
    fail(path, message(error));
  }

  const admin = createAdminClient();

  const { data: existing, error: lookupError } = await admin
    .from("coupons")
    .select("id")
    .eq("code", coupon.code)
    .maybeSingle();

  if (lookupError) {
    fail(path, "Unable to check coupon codes.");
  }

  if (existing) {
    fail(path, "This coupon code already exists.");
  }

  let discountId: string | null = null;

  if (coupon.active) {
    try {
      discountId = await createRemote(coupon);
    } catch (error) {
      console.error("Creem discount creation failed:", error);
      fail(path, message(error));
    }
  }

  const { data: created, error: insertError } = await admin
    .from("coupons")
    .insert({
      ...coupon,
      creem_discount_id: discountId,
      updated_at: new Date().toISOString(),
    })
    .select("id")
    .single();

  if (insertError || !created) {
    if (discountId) {
      try {
        await cleanupRemote(discountId);
      } catch (error) {
        fail(path, message(error));
      }
    }

    fail(
      path,
      insertError?.code === "23505"
        ? "This code already exists."
        : "Unable to save coupon."
    );
  }

  revalidatePath("/admin/coupons");

  redirect(`/admin/coupons/${created.id}/edit?created=1`);
}

export async function updateCoupon(
  couponId: string,
  formData: FormData
): Promise<void> {
  await requireAdmin();

  const path = `/admin/coupons/${couponId}/edit`;

  let updated: CouponData;
  let existing: StoredCoupon;

  try {
    updated = parseCoupon(formData);
    existing = await loadCoupon(couponId);
  } catch (error) {
    fail(path, message(error));
  }

  if (updated.active !== existing.active) {
    fail(path, "Use the coupon list to change status.");
  }

  if (
    existing.creem_discount_id &&
    termsChanged(existing, updated)
  ) {
    fail(
      path,
      "Disable this coupon before editing its discount rules."
    );
  }

  if (
    existing.active &&
    !existing.creem_discount_id &&
    termsChanged(existing, updated)
  ) {
    fail(
      path,
      "Synchronize or disable this coupon before editing its discount rules."
    );
  }

  const admin = createAdminClient();

  const { error } = await admin
    .from("coupons")
    .update({
      ...updated,
      updated_at: new Date().toISOString(),
    })
    .eq("id", couponId);

  if (error) {
    fail(
      path,
      error.code === "23505"
        ? "This code already exists."
        : "Unable to save coupon."
    );
  }

  revalidatePath("/admin/coupons");
  revalidatePath(path);

  redirect(`${path}?saved=1`);
}

export async function syncCoupon(
  couponId: string
): Promise<void> {
  await requireAdmin();

  const path = "/admin/coupons";

  let coupon: StoredCoupon;

  try {
    coupon = await loadCoupon(couponId);

    if (!coupon.active) {
      throw new Error(
        "Enable this coupon instead of synchronizing an inactive record."
      );
    }

    if (coupon.creem_discount_id) {
      throw new Error("This coupon already has a Creem mapping.");
    }
  } catch (error) {
    fail(path, message(error));
  }

  let discountId: string;

  try {
    discountId = await createRemote(coupon);
  } catch (error) {
    console.error("Coupon synchronization failed:", error);
    fail(path, message(error));
  }

  const admin = createAdminClient();

  const { data, error } = await admin
    .from("coupons")
    .update({
      creem_discount_id: discountId,
      updated_at: new Date().toISOString(),
    })
    .eq("id", couponId)
    .eq("active", true)
    .is("creem_discount_id", null)
    .select("id")
    .maybeSingle();

  if (error || !data) {
    try {
      await cleanupRemote(discountId);
    } catch (cleanupError) {
      fail(path, message(cleanupError));
    }

    fail(path, "Unable to save the Creem mapping.");
  }

  revalidatePath("/admin/coupons");
  revalidatePath(`/admin/coupons/${couponId}/edit`);

  redirect("/admin/coupons?success=synced");
}

export async function toggleCouponStatus(
  couponId: string,
  nextStatus: boolean
): Promise<void> {
  await requireAdmin();

  const path = "/admin/coupons";

  let coupon: StoredCoupon;

  try {
    coupon = await loadCoupon(couponId);
  } catch (error) {
    fail(path, message(error));
  }

  if (coupon.active === nextStatus) {
    redirect(path);
  }

  const admin = createAdminClient();

    try {
      await ensureNoActiveCheckouts(coupon.code);

        await deleteRemote(coupon.creem_discount_id);
      }
    } catch (error) {
      fail(path, message(error));
    }

      .from("coupons")
      .update({
        active: false,
        creem_discount_id: null,
        updated_at: new Date().toISOString(),
      })
      .eq("id", couponId)
      .eq("active", true)
      .select("id")
      .maybeSingle();

    if (error || !data) {
      fail(
        path,
        "Creem discount was processed but local status could not be updated. Review the coupon before retrying."
      );
    }
  } else {
    if (coupon.creem_discount_id) {
      fail(
        path,
        "This inactive coupon still has a Creem mapping. Resolve it before enabling."
      );
    }

    let discountId: string;

    try {
      discountId = await createRemote(coupon);
    } catch (error) {
      fail(path, message(error));
    }

      .from("coupons")

        active: true,
        creem_discount_id: discountId,
        updated_at: new Date().toISOString(),
      })
      .eq("id", couponId)
      .eq("active", false)
      .is("creem_discount_id", null)
      .select("id")

        await cleanupRemote(discountId);
      } catch (cleanupError) {
        fail(path, message(cleanupError));
      }

      fail(path, "Unable to enable coupon.");
    }
  }

  revalidatePath("/admin/coupons");
  revalidatePath(`/admin/coupons/${couponId}/edit`);

  redirect(
    `/admin/coupons?success=${nextStatus ? "enabled" : "disabled"}`
  );

import "server-only";

type RecordData = Record<string, unknown>;

function record(value: unknown): RecordData | null {
  return value !== null &&
    typeof value === "object" &&
    !Array.isArray(value)
    ? (value as RecordData)
    : null;
}

function requiredText(value: unknown, name: string): string {
  if (typeof value !== "string" || !value.trim()) {
    throw new Error(`Creem ${name} is missing.`);
  }

  return value.trim();
}

function nonnegativeInteger(
  value: unknown,
  name: string
): number {
  if (
    typeof value !== "number" ||
    !Number.isSafeInteger(value) ||
    value < 0
  ) {
    throw new Error(`Invalid Creem ${name}.`);
  }

  return value;
}

function baseUrl(): string {
  if (process.env.CREEM_ENVIRONMENT === "test") {
    return "https://test-api.creem.io";
  }

  if (process.env.CREEM_ENVIRONMENT === "production") {
    return "https://api.creem.io";
  }

  throw new Error("CREEM_ENVIRONMENT must be test or production.");
}

async function creemRequest(
  path: string,
  init: RequestInit = {}
): Promise<RecordData> {
  const key = process.env.CREEM_API_KEY;

  if (!key) {
    throw new Error("CREEM_API_KEY is not configured.");
  }

  const response = await fetch(`${baseUrl()}${path}`, {
    ...init,
    headers: {
      "x-api-key": key,
      Accept: "application/json",
      "Content-Type": "application/json",
      ...init.headers,
    },
    cache: "no-store",
    signal: AbortSignal.timeout(15000),
  });

  const payload: unknown =
    await response.json().catch(() => null);

  if (!response.ok) {
    console.error("Creem request failed:", {
      path,
      status: response.status,
    });

    throw new Error(
      `Creem API request failed (${response.status}).`
    );
  }

  const data = record(payload);

  if (!data) {
    throw new Error("Creem returned an invalid response.");
  }

  return data;
}

  name: string;
  code: string;
  type: "percentage" | "fixed";

function parseDiscount(data: RecordData): CreemDiscount {
  return {
    id: requiredText(data.id, "discount ID"),
    code: requiredText(data.code, "discount code"),
    status:
      typeof data.status === "string"
        ? data.status
        : "unknown",
    type:
      typeof data.type === "string"
        ? data.type
        : "unknown",
    percentage:
      typeof data.percentage === "number"
        ? data.percentage
        : null,
    amount:
      typeof data.amount === "number"
        ? data.amount
        : null,
    currency:
      typeof data.currency === "string"
        ? data.currency
        : null,
    expiryDate:
      typeof data.expiry_date === "string"
        ? data.expiry_date
        : null,
    maxRedemptions:
      typeof data.max_redemptions === "number"
        ? data.max_redemptions
        : null,
    redeemCount:
      typeof data.redeem_count === "number"
        ? data.redeem_count
        : 0,
    products: Array.isArray(data.applies_to_products)
      ? data.applies_to_products.filter(
          (id): id is string => typeof id === "string"
        )
      : [],
  };
}

export async function createCreemDiscount(
  input: CreateCreemDiscountInput
): Promise<CreemDiscount> {
  const code = input.code.trim().toUpperCase();
  const name = input.name.trim();
  const productIds = [...new Set(input.productIds)];

  if (!/^[A-Z0-9]{1,32}$/.test(code)) {
    throw new Error("Invalid discount code.");
  }

  if (!name || name.length > 200) {
    throw new Error("Invalid discount name.");
  }

  if (
    !productIds.length ||
    productIds.some(
      (id) => !/^prod_[A-Za-z0-9_-]+$/.test(id)
    )
  ) {
    throw new Error("Valid Creem products are required.");
  }

  if (
    input.maxRedemptions != null &&
    (!Number.isSafeInteger(input.maxRedemptions) ||
      input.maxRedemptions <= 0)
  ) {
    throw new Error("Invalid redemption limit.");
  }

  if (
    input.expiryDate &&
    !Number.isFinite(new Date(input.expiryDate).getTime())
  ) {
    throw new Error("Invalid discount expiry.");
  }

  if (input.type === "percentage") {
    if (
      input.percentage === undefined ||
      !Number.isInteger(input.percentage) ||
      input.percentage < 1 ||
      input.percentage > 100
    ) {
      throw new Error("Invalid percentage discount.");
    }
  } else {
    if (
      input.amount === undefined ||
      !Number.isSafeInteger(input.amount) ||
      input.amount <= 0 ||
      !input.currency ||
      !/^[A-Z]{3}$/.test(input.currency.toUpperCase())
    ) {
      throw new Error("Invalid fixed discount.");
    }
  }

  const payload = await creemRequest("/v1/discounts", {
    method: "POST",
    body: JSON.stringify({
      name,
      code,
      type: input.type,
      duration: "once",
      applies_to_products: productIds,
      ...(input.type === "percentage"
        ? { percentage: input.percentage }
        : {
            amount: input.amount,
            currency: input.currency?.toUpperCase(),
          }),
      ...(input.maxRedemptions != null
        ? { max_redemptions: input.maxRedemptions }
        : {}),
      ...(input.expiryDate
        ? {
            expiry_date: new Date(
              input.expiryDate
            ).toISOString(),
          }
        : {}),
    }),
  });

  const discount = parseDiscount(payload);

  if (
    !/^dis_[A-Za-z0-9_-]+$/.test(discount.id) ||
    discount.code.toUpperCase() !== code
  ) {
    throw new Error("Creem returned an unexpected discount.");
  }

  return discount;
}

export async function getCreemDiscountByCode(
  code: string
): Promise<CreemDiscount> {
  const normalized = code.trim().toUpperCase();

  if (!/^[A-Z0-9]{1,32}$/.test(normalized)) {
    throw new Error("Invalid discount code.");
  }

  const data = await creemRequest(
    `/v1/discounts?discount_code=${encodeURIComponent(
      normalized
    )}`
  );

  return parseDiscount(data);
}

export async function getCreemTransaction(
  transactionId: string
): Promise<CreemTransaction> {
  if (!/^tran_[A-Za-z0-9_-]+$/.test(transactionId)) {
    throw new Error("Invalid Creem transaction ID.");
  }

  const result = await creemRequest(
    `/v1/transactions?transaction_id=${encodeURIComponent(
      transactionId
    )}`
  );

  const id = requiredText(result.id, "transaction ID");

  if (id !== transactionId) {
    throw new Error("Creem transaction ID mismatch.");
  }

  return {
    id,
    orderId: requiredText(
      typeof result.order === "string"
        ? result.order
        : record(result.order)?.id,
      "transaction order"
    ),
    customerId:
      typeof result.customer === "string"
        ? result.customer
        : typeof record(result.customer)?.id === "string"
          ? (record(result.customer)?.id as string)
          : null,
    currency: requiredText(
      result.currency,
      "transaction currency"
    ).toUpperCase(),
    status: requiredText(
      result.status,
      "transaction status"
    ),
    type: requiredText(
      result.type,
      "transaction type"
    ),
    amount: nonnegativeInteger(
      result.amount,
      "amount"
    ),
    discountAmount:
      result.discount_amount == null
        ? 0
        : nonnegativeInteger(
            result.discount_amount,
            "discount amount"
          ),
    taxAmount:
      result.tax_amount == null
        ? 0
        : nonnegativeInteger(
            result.tax_amount,
            "tax amount"
          ),
    amountPaid: nonnegativeInteger(
      result.amount_paid,
      "amount paid"
    ),
  };
}

export async function createCreemCheckout(input: {
  productId: string;
  orderNumber: string;
  customerEmail: string;
  successUrl: string;
  metadata: Record<string, string>;
  discountCode?: string | null;
}): Promise<{
  id: string;
  checkoutUrl: string;
}> {
  if (!/^prod_[A-Za-z0-9_-]+$/.test(input.productId)) {
    throw new Error("Invalid Creem product ID.");
  }

  if (!input.orderNumber || !input.customerEmail) {
    throw new Error("Missing checkout information.");
  }

  const payload = await creemRequest("/v1/checkouts", {
    method: "POST",
    body: JSON.stringify({
      product_id: input.productId,
      request_id: input.orderNumber,
      units: 1,
      customer: {
        email: input.customerEmail,
      },
      success_url: input.successUrl,
      metadata: input.metadata,
      ...(input.discountCode
        ? { discount_code: input.discountCode }
        : {}),
    }),
  });

  const id = requiredText(payload.id, "checkout ID");
  const url = requiredText(
    payload.checkout_url,
    "checkout URL"
  );

  if (!/^ch_[A-Za-z0-9_-]+$/.test(id)) {
    throw new Error("Invalid Creem checkout ID.");
  }

  const checkoutUrl = new URL(url);

  if (
    checkoutUrl.protocol !== "https:" ||
    !(
      checkoutUrl.hostname === "creem.io" ||
      checkoutUrl.hostname.endsWith(".creem.io")
    )
  ) {
    throw new Error("Unexpected Creem checkout URL.");
  }

  return {
    id,
    checkoutUrl: checkoutUrl.toString(),
  };
}

import "server-only";

import { getCreemDiscountByCode } from "@/lib/creem/server";
import { createAdminClient } from "@/lib/supabase/admin";

export type ValidatedCoupon = {
  code: string;
  description: string | null;
  discountType: "percentage" | "flat";
  amount: number;
  currency: string | null;
  discountCents: number;
  totalCents: number;
};

type ValidateCheckoutCouponInput = {
  code: string;
  productId: string;
  creemProductId: string;
  productPriceCents: number;
  currency: string;
};

function isExpired(value: string | null): boolean {
  if (!value) {
    return false;
  }

  const time = new Date(value).getTime();

  return !Number.isFinite(time) || time <= Date.now();
}

export async function validateCheckoutCoupon(
  input: ValidateCheckoutCouponInput
): Promise<ValidatedCoupon> {
  const code = input.code.trim().toUpperCase();
  const currency = input.currency.trim().toUpperCase();

  if (!/^[A-Z0-9]{1,32}$/.test(code)) {
    throw new Error("Invalid coupon code.");
  }

  if (
    !input.productId ||
    !/^prod_[A-Za-z0-9_-]+$/.test(input.creemProductId)
  ) {
    throw new Error("Invalid product.");
  }

  if (
    !Number.isSafeInteger(input.productPriceCents) ||
    input.productPriceCents <= 0
  ) {
    throw new Error("Invalid product price.");
  }

  if (!/^[A-Z]{3}$/.test(currency)) {
    throw new Error("Invalid product currency.");
  }

  const admin = createAdminClient();

  const { data: coupon, error } = await admin
    .from("coupons")
    .select(
      "code,description,discount_type,amount,currency,active,expires_at,usage_limit,creem_discount_id"
    )
    .eq("code", code)
    .maybeSingle();

  if (error) {
    throw new Error("Unable to check coupon.");
  }

  if (
    !coupon ||
    !coupon.active ||
    !coupon.creem_discount_id
  ) {
    throw new Error("Coupon is not available.");
  }

  if (isExpired(coupon.expires_at)) {
    throw new Error("Coupon has expired.");
  }

  if (
    coupon.discount_type !== "percentage" &&
    coupon.discount_type !== "flat"
  ) {
    throw new Error("Unsupported coupon type.");
  }

  const amount = Number(coupon.amount);

  if (!Number.isSafeInteger(amount) || amount <= 0) {
    throw new Error("Invalid coupon value.");
  }

  let creem;

  try {
    creem = await getCreemDiscountByCode(code);
  } catch {
    throw new Error(
      "Unable to verify this coupon with Creem."
    );
  }

  if (
    creem.id !== coupon.creem_discount_id ||
    creem.code.toUpperCase() !== code ||
    creem.status !== "active"
  ) {
    throw new Error(
      "Coupon is not active or synchronized with Creem."
    );
  }

  if (
    !creem.products.includes(input.creemProductId)
  ) {
    throw new Error(
      "Coupon does not apply to this product."
    );
  }

  if (isExpired(creem.expiryDate)) {
    throw new Error("Coupon has expired in Creem.");
  }

  if (
    creem.maxRedemptions !== null &&
    (
      !Number.isSafeInteger(creem.maxRedemptions) ||
      creem.maxRedemptions <= 0
    )
  ) {
    throw new Error("Invalid Creem redemption limit.");
  }

  if (
    !Number.isSafeInteger(creem.redeemCount) ||
    creem.redeemCount < 0
  ) {
    throw new Error("Invalid Creem redemption count.");
  }

  if (
    creem.maxRedemptions !== null &&
    creem.redeemCount >= creem.maxRedemptions
  ) {
    throw new Error(
      "Coupon redemption limit has been reached."
    );
  }

  if (
    coupon.usage_limit !== null &&
    Number(coupon.usage_limit) !== creem.maxRedemptions
  ) {
    throw new Error(
      "Coupon usage limit is not synchronized with Creem."
    );
  }

  let discountCents: number;

  if (coupon.discount_type === "percentage") {
    if (
      amount > 100 ||
      creem.type !== "percentage" ||
      creem.percentage !== amount
    ) {
      throw new Error(
        "Coupon percentage does not match Creem."
      );
    }

    discountCents = Math.round(
      (input.productPriceCents * amount) / 100
    );
  } else {
    if (
      creem.type !== "fixed" ||
      creem.amount !== amount ||
      coupon.currency?.toUpperCase() !== currency ||
      creem.currency?.toUpperCase() !== currency
    ) {
      throw new Error(
        "Coupon amount or currency does not match Creem."
      );
    }

    discountCents = amount;
  }

  if (
    !Number.isSafeInteger(discountCents) ||
    discountCents <= 0
  ) {
    throw new Error("Invalid calculated discount.");
  }

  discountCents = Math.min(
    discountCents,
    input.productPriceCents
  );

  return {
    code,
    description: coupon.description,
    discountType: coupon.discount_type,
    amount,
    currency: coupon.currency,
    discountCents,
    totalCents: input.productPriceCents - discountCents,
  };
}
